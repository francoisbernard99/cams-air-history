# Ce fichier ne s'exporte pas seul. Il est inclus par le rapport et par la
# présentation, pour que les deux citent exactement les mêmes chiffres.
#
# Le principe : aucun nombre n'est recopié à la main dans les deux documents.
# Un chiffre recopié devient faux le jour où la donnée bouge, et personne ne
# s'en aperçoit. Ici, réévaluer les blocs (C-c C-v b) met tout à jour.
#
# Prérequis, depuis la racine du dépôt :
#   git worktree add archive data          # les données
#   .venv/bin/python -m warehouse build --data-dir archive/data
#
# Puis dans Emacs : C-c C-v b pour tout réévaluer avant export.
#
#
# DEUX POINTS À NE PAS DÉFAIRE
#
# 1. Chaque bloc porte son en-tête complet (:noweb, :exports, :eval), alors
#    qu'une ligne #+PROPERTY en tête de fichier dirait la même chose en une
#    fois. #+INCLUDE n'importe QUE le contenu du sous-arbre : les mots-clés du
#    fichier restent derrière. Un bloc inclus ailleurs perdrait donc ses
#    réglages et se retrouverait imprimé tel quel dans le document.
#
# 2. Aucun bloc n'utilise :session, et c'est délibéré. En mode session,
#    org-babel envoie le corps ligne à ligne à un interpréteur vivant, où un
#    `return` au niveau supérieur est une erreur de syntaxe — silencieusement
#    avalée, laissant un tableau vide. Sans session, org-babel enveloppe le
#    corps dans une fonction et `return` fonctionne. Le code commun est partagé
#    par noweb (<<preparation>>), qui l'insère dans chaque bloc à l'évaluation.
#    Chaque bloc est donc autonome et reproductible isolément.

* Préparation                                                      :noexport:

#+NAME: preparation
#+BEGIN_SRC python :results none :exports none :eval never-export
import duckdb, pathlib

RACINE = pathlib.Path("../..").resolve()
con = duckdb.connect(str(RACINE / "air.duckdb"), read_only=True)

# Noms lisibles pour le document. Une espèce absente de ce tableau garde sa clé,
# donc en activer une nouvelle dans le collecteur ne casse rien ici.
ESPECES = {
    "pm2_5": "PM2,5", "pm10": "PM10",
    "nitrogen_dioxide": "NO₂", "ozone": "O₃",
    "sulphur_dioxide": "SO₂", "carbon_monoxide": "CO",
}

def cellule(valeur):
    """Org-babel ne convertit que les nombres et les chaînes.

    Une date Python lui arrive comme un objet qu'il éclate en deux cellules,
    ce qui décale silencieusement toute la ligne. On convertit donc ici tout
    ce qui n'est pas un nombre, et on traduit les noms d'espèces au passage.
    """
    if valeur is None:
        return ""
    if isinstance(valeur, bool):
        return str(valeur)
    if isinstance(valeur, (int, float)):
        return valeur
    return ESPECES.get(str(valeur), str(valeur))

def table(sql):
    """Une requête, un tableau Org : en-têtes, filet, puis les lignes."""
    curseur = con.execute(sql)
    entetes = [d[0] for d in curseur.description]
    lignes = [[cellule(v) for v in ligne] for ligne in curseur.fetchall()]
    return [entetes, None] + lignes
#+END_SRC

* Les écarts les plus forts à la normale

# Le tableau qui porte le résultat central : trois espèces indépendantes
# désignent la même journée au même endroit.

#+NAME: tab-anomalies
#+BEGIN_SRC python :noweb yes :results value table :exports results :eval never-export
<<preparation>>
return table("""
WITH spread AS (
  SELECT species, median(mean_value) AS mediane,
         quantile_cont(mean_value, 0.90) AS chargee
  FROM daily WHERE hours = 24 GROUP BY species)
SELECT d.species AS "Espèce", d.day AS "Journée", d.site AS "Point",
       round(d.mean_value,1) AS "Moyenne",
       round(s.mediane,2)    AS "Médiane de l'espèce",
       round((d.mean_value - s.mediane)/nullif(s.chargee - s.mediane,0),1) AS "Écart"
FROM daily d JOIN spread s USING (species)
WHERE d.hours = 24
ORDER BY (d.mean_value - s.mediane)/nullif(s.chargee - s.mediane,0) DESC
LIMIT 5""")
#+END_SRC

* Modèle contre mesure au sol, épisode de Bordeaux

# Nécessite le réseau : lit le fichier Parquet de l'AEE directement par URL.
# Station FR31002, fond urbain, à 4,5 km du point CAMS.

#+NAME: tab-validation
#+BEGIN_SRC python :noweb yes :results value table :exports results :eval never-export
<<preparation>>
con.execute("INSTALL httpfs; LOAD httpfs;")
URL = ("https://eeadmz1batchservice02.blob.core.windows.net/"
       "airquality-p/FR/SPO-FR31002_6001.parquet")
con.execute(f"CREATE OR REPLACE TEMP VIEW station AS SELECT * FROM read_parquet('{URL}')")
return table("""
WITH mesure AS (
  SELECT Start AS h, CAST(Value AS DOUBLE) AS mesure
  FROM station WHERE Validity >= 1
    AND CAST(Start AS DATE) BETWEEN DATE '2026-07-25' AND DATE '2026-07-27'),
modele AS (
  SELECT measured_at AS h, value AS cams FROM readings
  WHERE site='Bordeaux' AND species='pm2_5'
    AND CAST(measured_at AS DATE) BETWEEN DATE '2026-07-25' AND DATE '2026-07-27')
SELECT CAST(m.h AS DATE) AS "Journée", count(*) AS "Heures",
       round(avg(s.mesure),1) AS "Station", round(avg(m.cams),1) AS "CAMS",
       round(avg(m.cams - s.mesure),1) AS "Biais",
       round(max(s.mesure),1) AS "Pointe station",
       round(max(m.cams),1)   AS "Pointe CAMS"
FROM modele m JOIN mesure s USING (h) GROUP BY 1 ORDER BY 1""")
#+END_SRC
